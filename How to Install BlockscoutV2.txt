# Generating and Placing SSL Certificates
mkdir ssl
mkdir ssl/explorer
mkdir ssl/stats
mkdir ssl/visualizer

# Create SSL using certbot

certbot certonly
>> vrixa.io

certbot certonly
>> stats.vrixa.io

certbot certonly
>> visualizer.vrixa.io

Note: We run the command for each domain one by one

# Copy the certificates

cd ssl/explorer
sudo cp /etc/letsencrypt/live/vrixa.io/fullchain.pem .
sudo cp /etc/letsencrypt/live/vrixa.io/privkey.pem .

cd ssl/stats
sudo cp /etc/letsencrypt/live/stats.vrixa.io/fullchain.pem .
sudo cp /etc/letsencrypt/live/stats.vrixa.io/privkey.pem .

cd ssl/visualizer
